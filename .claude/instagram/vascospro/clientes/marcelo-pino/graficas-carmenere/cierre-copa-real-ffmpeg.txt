# Composición del cierre sobre el video de la copa (1920x1080, 3,9 s) enviado por Jorge.
# Escala a 1500 de alto, recorte 1080 de ancho siguiendo la copa (deja fuera el texto
# "Día Nacional" del clip original), cámara lenta x1,3, congela el último cuadro y pone
# encima los textos con alfa (E_overlay) más un degradé para ocultar el borde del video.
ffmpeg -i copa.mp4 -i fade.png -framerate 30 -i E_overlay/f%04d.png -filter_complex \
 "[0:v]scale=-2:1500,crop=1080:1500:x='(0.39-0.08*t/3.9)*iw-540':y=0,setpts=PTS*1.3,fps=30,tpad=stop_mode=clone:stop_duration=2[v];\
  color=c=0x13060a:s=1080x1920:r=30:d=5[c];[c][v]overlay=0:-60[bg];[bg][1:v]overlay=0:1090[bg2];[bg2][2:v]overlay=format=auto,format=yuv420p" \
 -t 5 -an -c:v libx264 -crf 17 -movflags +faststart cierre_copa_real.mp4
